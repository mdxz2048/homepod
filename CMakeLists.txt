cmake_minimum_required(VERSION 3.16)
project(HomePodFlowCore VERSION 0.1.0 LANGUAGES C CXX)

option(FLOW_BUILD_EXAMPLES "Build offline examples (no audio/network)" ON)
option(FLOW_BUILD_TESTS "Build independent core contract tests" ON)

add_subdirectory(core)
if(FLOW_BUILD_EXAMPLES)
  add_executable(flow_offline_demo examples/offline_demo.cpp)
  target_link_libraries(flow_offline_demo PRIVATE flow_core)
endif()
if(FLOW_BUILD_TESTS)
  include(CTest)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/bridge_c_contract.c")
    add_executable(flow_bridge_c_tests tests/bridge_c_contract.c)
    target_include_directories(flow_bridge_c_tests PRIVATE core/include)
    target_link_libraries(flow_bridge_c_tests PRIVATE flow_bridge)
    set_target_properties(flow_bridge_c_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    add_test(NAME flow_bridge_c_tests COMMAND flow_bridge_c_tests)
  endif()
endif()
