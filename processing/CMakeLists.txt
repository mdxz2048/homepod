cmake_minimum_required(VERSION 3.16)
project(flow_processing LANGUAGES CXX)
add_library(flow_processing SHARED src/processing.cpp)
target_compile_features(flow_processing PUBLIC cxx_std_17)
target_include_directories(flow_processing PUBLIC include)
target_compile_definitions(flow_processing PRIVATE FLOW_PROCESSING_BUILD)
set_target_properties(flow_processing PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(FLOW_PROCESSING_TESTS "Build DSP behavior checks" ON)
if(FLOW_PROCESSING_TESTS)
  enable_testing()
  add_executable(flow_processing_checks ../tests/processing_checks.cpp)
  target_link_libraries(flow_processing_checks PRIVATE flow_processing)
  add_test(NAME processing_behavior COMMAND flow_processing_checks)
endif()
