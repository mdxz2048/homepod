add_library(flow_core STATIC src/core.cpp)
add_library(flow::core ALIAS flow_core)
target_include_directories(flow_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(flow_core PUBLIC cxx_std_17)
set_target_properties(flow_core PROPERTIES CXX_EXTENSIONS OFF)
set_target_properties(flow_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(flow_bridge SHARED src/bridge.cpp)
target_link_libraries(flow_bridge PRIVATE flow_core)
target_compile_definitions(flow_bridge PRIVATE FLOW_BRIDGE_BUILD)
target_compile_features(flow_bridge PRIVATE cxx_std_17)
set_target_properties(flow_bridge PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(flow_core PRIVATE /W4)
else()
  target_compile_options(flow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
